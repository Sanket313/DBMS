DELIMITER $$

CREATE PROCEDURE IncreaseSalary()
BEGIN
    DECLARE done INT DEFAULT 0;
    DECLARE emp_id INT;

    DECLARE emp_cursor CURSOR FOR
        SELECT id FROM Employee;

    DECLARE CONTINUE HANDLER FOR NOT FOUND
        SET done = 1;

    OPEN emp_cursor;

    read_loop: LOOP
        FETCH emp_cursor INTO emp_id;

        IF done = 1 THEN
            LEAVE read_loop;
        END IF;

        UPDATE Employee
        SET salary = salary + 1000
        WHERE id = emp_id;
    END LOOP;

    CLOSE emp_cursor;
END$$

DELIMITER ;

CALL IncreaseSalary();
