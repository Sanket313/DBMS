CREATE TABLE Books (
    book_id INT PRIMARY KEY,
    book_name VARCHAR(100) NOT NULL,
    author_name VARCHAR(100),
    category VARCHAR(50)
);

CREATE TABLE Issue (
    issue_id INT PRIMARY KEY,
    book_id INT,
    student_name VARCHAR(100),
    issue_date DATE,
    return_date DATE,
    FOREIGN KEY (book_id) REFERENCES Books(book_id)
);

SELECT 
    Books.book_id,
    Books.book_name,
    Issue.student_name,
    Issue.issue_date
FROM Books
INNER JOIN Issue
ON Books.book_id = Issue.book_id;

SELECT 
    Books.book_id,
    Books.book_name,
    Issue.student_name,
    Issue.issue_date
FROM Books
LEFT JOIN Issue
ON Books.book_id = Issue.book_id;

SELECT 
    Books.book_id,
    Books.book_name,
    Issue.student_name,
    Issue.issue_date
FROM Books
RIGHT JOIN Issue
ON Books.book_id = Issue.book_id;

CREATE VIEW Current_Issued_Books AS
SELECT 
    Books.book_id,
    Books.book_name,
    Issue.student_name,
    Issue.issue_date
FROM Books
INNER JOIN Issue
ON Books.book_id = Issue.book_id
WHERE Issue.return_date IS NULL;
